"OCR" : "PDF text"); card.appendChild(method); const line = document.createElement("hr"); card.appendChild(line); function createField(labelText, value, id) { const wrapper = document.createElement("div"); wrapper.appendChild(label); wrapper.appendChild(input); card.appendChild(wrapper); return input; } const typeInput = createField( "Document type", type, "editType" ); const nameInput = createField( "Name", name, "editName" ); const professionInput = createField( "Profession", profession, "editProfession" ); const locationInput = createField( "Location", location, "editLocation" ); const emailInput = createField( "Email", email, "editEmail" );
